package model;

/**
 *
 * @author ihabh
 */
public class Recipe {
    private String nama;
    private String kategori;
    private int waktu;

    public Recipe(String nama, String kategori, int waktu) {
        this.nama = nama;
        this.kategori = kategori;
        this.waktu = waktu;
    }
    // Getter & Setter
    public String getNama() { return nama; }
    public String getKategori() { return kategori; }
    public int getWaktu() { return waktu; }
}

4. proses.jsp
"
<%@page import="java.sql.*, config.DBConfig" %>
<%
    String aksi = request.getParameter("aksi");
    Connection conn = DBConfig.getConnection();

    if("create".equals(aksi)){
        String nama = request.getParameter("nama");
        // CEK DUPLIKAT
        PreparedStatement cek = conn.prepareStatement("SELECT * FROM recipes WHERE nama_resep=?");
        cek.setString(1, nama);
        if(cek.executeQuery().next()){
            out.println("<script>alert('Nama resep sudah ada!'); window.location='tambah.jsp';</script>");
        } else {
            PreparedStatement ps = conn.prepareStatement("INSERT INTO recipes (nama_resep, kategori, waktu_masak) VALUES (?,?,?)");
            ps.setString(1, nama);
            ps.setString(2, request.getParameter("kategori"));
            ps.setInt(3, Integer.parseInt(request.getParameter("waktu")));
            ps.executeUpdate();
            response.sendRedirect("index.jsp");
        }
    } 
    else if("update".equals(aksi)){
        PreparedStatement ps = conn.prepareStatement("UPDATE recipes SET nama_resep=?, kategori=?, waktu_masak=? WHERE nama_resep=?");
        ps.setString(1, request.getParameter("nama"));
        ps.setString(2, request.getParameter("kategori"));
        ps.setInt(3, Integer.parseInt(request.getParameter("waktu")));
        ps.setString(4, request.getParameter("nama_lama"));
        ps.executeUpdate();
        response.sendRedirect("index.jsp");
    }
    else if("delete".equals(aksi)){
        PreparedStatement ps = conn.prepareStatement("DELETE FROM recipes WHERE nama_resep=?");
        ps.setString(1, request.getParameter("nama"));
        ps.executeUpdate();
        response.sendRedirect("index.jsp");
    }
%>
