3. app/Http/Controllers/MahasiswaController.php
```
<?php

namespace App\Http\Controllers;

use Illuminate\Http\Request;

class MahasiswaController extends Controller
{
    // ... (Isi file controller sama seperti sebelumnya) ...
    public function index(Request $request)
    {
        $mahasiswa = $request->session()->get('mahasiswa', []);
        $totalMahasiswa = count($mahasiswa);
        return view('mahasiswa.index', compact('mahasiswa', 'totalMahasiswa'));
    }

    public function create()
    {
        return view('mahasiswa.create');
    }

    public function store(Request $request)
    {
        // Validasi input sesuai ketentuan [cite: 16-18]
        $validated = $request->validate([
            'nama' => 'required', // Nama tidak boleh kosong
            'nim' => 'required|numeric|digits:12', // NIM harus berupa angka dengan panjang tepat 12 digit
            'nilai_pw' => 'required|numeric|between:0,100', // Nilai harus angka 0-100
            'nilai_ppm' => 'required|numeric|between:0,100',
            'nilai_mp' => 'required|numeric|between:0,100',
        ]);

        // Persiapan data baru
        $dataMahasiswaBaru = [
            'nama' => $validated['nama'],
            'nim' => $validated['nim'],
            'nilai' => [
                'Pemrograman Web' => $validated['nilai_pw'],
                'Pemrograman Perangkat Bergerak' => $validated['nilai_ppm'],
                'Manajemen Proyek' => $validated['nilai_mp'],
            ],
        ];

        // Simpan data ke session array [cite: 55]
        $mahasiswa = $request->session()->get('mahasiswa', []);
        $mahasiswa[] = $dataMahasiswaBaru;
        $request->session()->put('mahasiswa', $mahasiswa);

        // Redirect dengan pesan sukses
        return redirect()->route('mahasiswa.index')->with('success', 'Data berhasil disimpan!');
    }
}
```
