<?php

namespace App\Http\Controllers;

use App\Models\Makanan; // Import Model Makanan
use Illuminate\Http\Request;

class MakananController extends Controller
{
    /**
     * Menampilkan daftar semua makanan (Read).
     */
    public function index()
    {
        // Ambil semua data makanan
        $makanans = Makanan::all();
        
        // Kirim data ke view 'makanans.index'
        return view('makanans.index', ['makanans' => $makanans]);
    }

    /**
     * Menampilkan form untuk menambah makanan baru (Create - view).
     */
    public function create()
    {
        // Kirim data untuk mengisi form.blade.php
        return view('makanans.form', [
            'title' => 'Tambah',
            'makanan' => new Makanan(),
            'route' => route('makanans.store'),
            'method' => 'POST',
        ]);
    }

    /**
     * Menyimpan makanan baru ke database (Create - store).
     */
    public function store(Request $request)
    {
        // Validasi input
        $validated = $request->validate([
            'nama_makanan' => 'required|min:3', 
            // Harga dan Deskripsi menggunakan string (varchar) di migrasi
            'harga' => 'required|string|min:1', 
            'deskripsi' => 'required|string|min:5',
            'stock' => 'required|integer|min:0', // Stock harus integer
        ]);
        
        // Simpan data ke database
        Makanan::create($validated);
        
        // Redirect dengan pesan sukses (Bootstrap alert)
        return redirect()->route('makanans.index')->with('success', 'Data berhasil ditambahkan'); [cite: 61-62]
    }
}
